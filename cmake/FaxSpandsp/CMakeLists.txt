cmake_minimum_required(VERSION 3.21)
project(StelSpanDSP LANGUAGES C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
option(BUILD_SHARED_LIBS "Build the portable fax DLL" ON)
set(SPANDSP_SOURCE "" CACHE PATH "Checksum-verified SpanDSP 0.0.6 source")
find_package(TIFF REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(src "${SPANDSP_SOURCE}/src")
file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/include")
if(MSVC)
  file(READ "${src}/msvc/config.h" config)
  string(REPLACE "#define snprintf _snprintf" "/* Modern MSVC supplies snprintf. */" config "${config}")
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/config.h" "${config}\n#define HAVE_STDBOOL_H 1\n")
  foreach(file unistd.h tgmath.h sys/time.h)
    configure_file("${src}/msvc/${file}" "include/${file}" COPYONLY)
  endforeach()
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/include/getopt.h" "#include <unistd.h>\nextern int optind, opterr, optopt;\n")
  add_compile_definitions(_WIN32_WINNT=0x0601 WINVER=0x0601 _CRT_SECURE_NO_WARNINGS __inline__=__inline)
else()
  # Used to verify the same fax source subset on Linux. Configure only generates
  # feature headers; CMake builds the library and modem-filter generators.
  execute_process(COMMAND "${SPANDSP_SOURCE}/configure" --disable-shared
    WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}" COMMAND_ERROR_IS_FATAL ANY)
  configure_file("${CMAKE_CURRENT_BINARY_DIR}/src/config.h" include/config.h COPYONLY)
endif()
add_executable(stel_modem_filter "${src}/make_modem_filter.c" "${src}/filter_tools.c")
if(MSVC)
  target_sources(stel_modem_filter PRIVATE "${src}/msvc/getopt.c")
else()
  target_link_libraries(stel_modem_filter PRIVATE m)
endif()
target_include_directories(stel_modem_filter PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/include" "${src}")
target_compile_definitions(stel_modem_filter PRIVATE HAVE_CONFIG_H)
set(sources async bit_operations complex_vector_float crc dds_float dds_int fax fax_modems fsk hdlc logging
    modem_connect_tones power_meter silence_gen t30 t30_api t30_logging t35 t38_core t38_terminal
    t4_rx t4_tx timezone v17rx v17tx v27ter_rx v27ter_tx v29rx v29tx vector_float)
list(TRANSFORM sources PREPEND "${src}/")
list(TRANSFORM sources APPEND ".c")
add_custom_command(OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/tables.ready"
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py" "${src}"
    "${CMAKE_CURRENT_BINARY_DIR}/include" "$<TARGET_FILE:stel_modem_filter>"
  COMMAND "${CMAKE_COMMAND}" -E touch "${CMAKE_CURRENT_BINARY_DIR}/tables.ready"
  DEPENDS stel_modem_filter "${CMAKE_CURRENT_SOURCE_DIR}/prepare.py")
add_library(stelspandsp ${sources} "${CMAKE_CURRENT_BINARY_DIR}/tables.ready")
if(NOT BUILD_SHARED_LIBS)
  target_compile_definitions(stelspandsp PRIVATE SPANDSP_STATIC)
endif()
if(MSVC)
  target_sources(stelspandsp PRIVATE "${src}/msvc/gettimeofday.c")
  target_sources(stelspandsp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/tiff-utf8.c")
  target_compile_definitions(stelspandsp PRIVATE TIFFOpen=stel_fax_tiff_open_utf8)
  target_link_libraries(stelspandsp PRIVATE ws2_32)
else()
  target_link_libraries(stelspandsp PRIVATE m)
endif()
target_compile_definitions(stelspandsp PRIVATE HAVE_CONFIG_H LIBSPANDSP_EXPORTS)
target_include_directories(stelspandsp PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/include" "${src}")
target_link_libraries(stelspandsp PRIVATE TIFF::TIFF)
install(TARGETS stelspandsp RUNTIME DESTINATION bin LIBRARY DESTINATION lib ARCHIVE DESTINATION lib)
install(DIRECTORY "${src}/spandsp" DESTINATION include FILES_MATCHING PATTERN "*.h")
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/include/spandsp.h" DESTINATION include)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/include/spandsp/fast_convert.h" DESTINATION include/spandsp)

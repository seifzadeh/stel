cmake_minimum_required(VERSION 3.21)
project(Stel VERSION 0.6.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
set(STEL_QT_MAJOR "5" CACHE STRING "Qt major version (5 for Windows 7/8.1)")
set(STEL_LINPHONE_ROOT "" CACHE PATH "Linphone SDK prefix; empty uses installed SDK")
option(STEL_DISTRO_LINPHONE "Use direct shared liblinphone discovery (Debian packaging workaround)" OFF)
option(STEL_STATIC "Require static Qt and static Linphone SDK" OFF)
option(STEL_BUILD_TESTS "Build tests" ON)
find_package(Qt${STEL_QT_MAJOR} REQUIRED COMPONENTS Widgets Network)
find_package(Qt${STEL_QT_MAJOR}Keychain CONFIG REQUIRED)
add_executable(stel_brand_assets EXCLUDE_FROM_ALL scripts/render-brand.cpp)
set_target_properties(stel_brand_assets PROPERTIES AUTOMOC OFF AUTORCC OFF)
target_link_libraries(stel_brand_assets PRIVATE Qt${STEL_QT_MAJOR}::Widgets)
if(STEL_DISTRO_LINPHONE)
  if(STEL_STATIC)
    message(FATAL_ERROR "Static builds require SDK CMake targets and their transitive dependencies.")
  endif()
  find_path(STEL_LINPHONE_INCLUDE linphone/linphonecore.h HINTS "${STEL_LINPHONE_ROOT}/include" REQUIRED)
  find_library(STEL_LINPHONE_LIBRARY NAMES linphone HINTS "${STEL_LINPHONE_ROOT}/lib" "${STEL_LINPHONE_ROOT}/lib/x86_64-linux-gnu" REQUIRED)
  find_library(STEL_BELR_LIBRARY NAMES belr HINTS "${STEL_LINPHONE_ROOT}/lib/x86_64-linux-gnu" REQUIRED)
  find_library(STEL_BCTOOLBOX_LIBRARY NAMES bctoolbox HINTS "${STEL_LINPHONE_ROOT}/lib/x86_64-linux-gnu" REQUIRED)
  add_library(stel_linphone SHARED IMPORTED)
  set_target_properties(stel_linphone PROPERTIES IMPORTED_LOCATION "${STEL_LINPHONE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${STEL_LINPHONE_INCLUDE}")
  set(STEL_ENGINE_TARGET stel_linphone)
else()
  if(STEL_LINPHONE_ROOT)
    list(PREPEND CMAKE_PREFIX_PATH "${STEL_LINPHONE_ROOT}")
  endif()
  find_package(LibLinphone CONFIG REQUIRED)
  set(STEL_ENGINE_TARGET ${LibLinphone_TARGET})
  if(NOT STEL_ENGINE_TARGET)
    set(STEL_ENGINE_TARGET liblinphone)
  endif()
endif()
if(STEL_STATIC)
  foreach(t Qt${STEL_QT_MAJOR}::Core qt${STEL_QT_MAJOR}keychain ${STEL_ENGINE_TARGET})
    get_target_property(kind ${t} TYPE)
    if(NOT kind STREQUAL "STATIC_LIBRARY")
      message(FATAL_ERROR "${t} is ${kind}; supply a static SDK before enabling STEL_STATIC.")
    endif()
  endforeach()
endif()
if(MSVC)
  add_compile_options(/W4 /utf-8)
  if(STEL_STATIC)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  endif()
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
add_library(stel_core src/model.cpp src/sipengine.cpp src/dummydevices.cpp src/secretstore.cpp)
option(STEL_AV1_ADDON "Build matching MS2 5.3 AV1 filters (use OFF when SDK already provides AV1)" OFF)
option(STEL_LEGACY_VIDEO_ADDON "Build MS2 5.3 H.263/MPEG-4 filters adapted for FFmpeg 7" OFF)
if(STEL_LEGACY_VIDEO_ADDON)
  find_path(STEL_FFMPEG_INCLUDE libavcodec/avcodec.h PATH_SUFFIXES x86_64-linux-gnu REQUIRED)
  find_library(STEL_FFMPEG_LIBRARY NAMES avcodec REQUIRED)
  find_library(STEL_AVUTIL_LIBRARY NAMES avutil REQUIRED)
  find_library(STEL_SWSCALE_LIBRARY NAMES swscale REQUIRED)
  target_sources(stel_core PRIVATE third_party/ms2-legacy/videoenc.c third_party/ms2-legacy/videodec.c)
  target_include_directories(stel_core PRIVATE ${STEL_FFMPEG_INCLUDE})
  target_link_libraries(stel_core PUBLIC ${STEL_FFMPEG_LIBRARY} ${STEL_AVUTIL_LIBRARY} ${STEL_SWSCALE_LIBRARY})
  target_compile_definitions(stel_core PRIVATE STEL_LEGACY_VIDEO_ADDON)
endif()
if(STEL_AV1_ADDON)
  find_path(STEL_AOM_INCLUDE aom/aom_encoder.h REQUIRED)
  find_path(STEL_DAV1D_INCLUDE dav1d/dav1d.h REQUIRED)
  find_library(STEL_AOM_LIBRARY NAMES aom REQUIRED)
  find_library(STEL_DAV1D_LIBRARY NAMES dav1d REQUIRED)
  set(av1_root ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ms2-av1/src)
  file(GLOB_RECURSE av1_sources CONFIGURE_DEPENDS ${av1_root}/videofilters/av1/*.cpp ${av1_root}/videofilters/av1/*.c ${av1_root}/utils/filter-wrapper/*.cpp)
  target_sources(stel_core PRIVATE ${av1_sources})
  target_include_directories(stel_core PRIVATE ${av1_root}/voip ${av1_root}/utils ${STEL_AOM_INCLUDE} ${STEL_DAV1D_INCLUDE})
  target_link_libraries(stel_core PUBLIC ${STEL_AOM_LIBRARY} ${STEL_DAV1D_LIBRARY})
  target_compile_definitions(stel_core PRIVATE STEL_AV1_ADDON)
endif()
target_include_directories(stel_core PUBLIC src)
target_link_libraries(stel_core PUBLIC Qt${STEL_QT_MAJOR}::Network Qt${STEL_QT_MAJOR}::Gui qt${STEL_QT_MAJOR}keychain ${STEL_ENGINE_TARGET})
if(STEL_DISTRO_LINPHONE)
  find_library(STEL_MS_LIBRARY NAMES mediastreamer2 HINTS "${STEL_LINPHONE_ROOT}/lib/x86_64-linux-gnu" REQUIRED)
  find_library(STEL_ORTP_LIBRARY NAMES ortp HINTS "${STEL_LINPHONE_ROOT}/lib/x86_64-linux-gnu" REQUIRED)
  target_link_libraries(stel_core PUBLIC ${STEL_BELR_LIBRARY} ${STEL_BCTOOLBOX_LIBRARY} ${STEL_MS_LIBRARY} ${STEL_ORTP_LIBRARY})
else()
  find_package(Belr CONFIG REQUIRED)
  target_link_libraries(stel_core PUBLIC ${Belr_TARGET})
endif()
target_compile_definitions(stel_core PUBLIC STEL_VERSION="${PROJECT_VERSION}")
add_library(stel_ui src/mainwindow.cpp src/calldialog.cpp src/videoview.cpp src/accountdialog.cpp src/settingswidget.cpp src/codecselector.cpp src/branding.cpp src/branding.qrc)
target_link_libraries(stel_ui PUBLIC stel_core Qt${STEL_QT_MAJOR}::Widgets)
add_executable(stel WIN32 src/main.cpp)
target_link_libraries(stel PRIVATE stel_ui)
option(STEL_PORTABLE_LAUNCHER "Build the native single-file distribution launcher" ON)
if(STEL_PORTABLE_LAUNCHER)
  add_executable(stel_portable_launcher WIN32 packaging/portable-launcher.cpp)
  set_target_properties(stel_portable_launcher PROPERTIES AUTOMOC OFF AUTORCC OFF)
  if(WIN32)
    target_compile_definitions(stel_portable_launcher PRIVATE _WIN32_WINNT=0x0601 WINVER=0x0601)
    target_link_libraries(stel_portable_launcher PRIVATE shell32 user32)
    if(MSVC)
      set_property(TARGET stel_portable_launcher PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endif()
  elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(ZLIB_USE_STATIC_LIBS ON)
    find_package(ZLIB REQUIRED)
    target_link_libraries(stel_portable_launcher PRIVATE ZLIB::ZLIB)
    target_link_options(stel_portable_launcher PRIVATE -static -no-pie -s)
  endif()
endif()
if(WIN32)
  enable_language(RC)
  configure_file(packaging/stel.rc.in generated/stel.rc @ONLY)
  target_sources(stel PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/stel.rc")
  target_include_directories(stel PRIVATE packaging)
  if(TARGET stel_portable_launcher)
    target_sources(stel_portable_launcher PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/generated/stel.rc")
    target_include_directories(stel_portable_launcher PRIVATE packaging)
  endif()
  target_compile_definitions(stel_core PUBLIC _WIN32_WINNT=0x0601 WINVER=0x0601)
endif()
include(GNUInstallDirs)
install(TARGETS stel RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
install(FILES packaging/stel.desktop DESTINATION share/applications)
install(FILES packaging/stel.svg DESTINATION share/icons/hicolor/scalable/apps)
foreach(size 16 24 32 48 64 128 256 512)
  install(FILES packaging/icons/stel-${size}.png DESTINATION share/icons/hicolor/${size}x${size}/apps RENAME stel.png)
endforeach()
install(FILES README.fa.md LICENSE THIRD_PARTY.md DESTINATION share/doc/stel)

  set_tests_properties(core PROPERTIES TIMEOUT 240)
  add_executable(stel_probe tests/probe.cpp)
  target_link_libraries(stel_probe PRIVATE stel_core)
  add_executable(stel_ui_tests tests/test_ui.cpp)
  target_link_libraries(stel_ui_tests PRIVATE stel_ui Qt${STEL_QT_MAJOR}::Test)
  add_executable(stel_pbx_check EXCLUDE_FROM_ALL tests/pbx_check.cpp)
  target_link_libraries(stel_pbx_check PRIVATE stel_ui Qt${STEL_QT_MAJOR}::Test)
  add_test(NAME settings_ui COMMAND stel_ui_tests)
  set_tests_properties(settings_ui PROPERTIES TIMEOUT 180 ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME tls_registration COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/test_tls.py $<TARGET_FILE:stel_probe>)
    set_tests_properties(tls_registration PROPERTIES TIMEOUT 90)
  endif()
endif()
